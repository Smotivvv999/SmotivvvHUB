-- ============================================
-- SETTINGS
-- ============================================
local AutoSapi = false
local KEY_DELAY = 0.001

local FARM_POSITION = Vector3.new(-3760.887939, -77.642609, -23892.484375)
local SEARCH_RADIUS = 300

-- ============================================
-- UI PARAGRAPH
-- ============================================
local Paragraph = Tab:CreateParagraph({
    Title = "STATUS TELEPORT",
    Content = "Menunggu aksi..."
})

local function UpdateParagraph(text)
    Paragraph:Set({
        Title = "STATUS TELEPORT",
        Content = text
    })
end

-- Throttled UI update (max 6-7x per detik)
local lastUI = ""
local lastUIUpdate = 0
local function SafeUI(text)
    if text == lastUI then return end
    lastUI = text
    local now = tick()
    if now - lastUIUpdate < 0.15 then return end
    lastUIUpdate = now
    UpdateParagraph(text)
end

-- ============================================
-- PRESS KEY
-- ============================================
local function pressKey(key)
    VirtualInputManager:SendKeyEvent(true, key, false, game)
    VirtualInputManager:SendKeyEvent(false, key, false, game)
end

-- ============================================
-- CLEAN TEXT
-- ============================================
local function cleanText(text)
    text = tostring(text)
    text = text:gsub("[\r\n]+", ""):gsub("%s+", "")
    return text
end

-- ============================================
-- TELEPORT FARM
-- ============================================
local function teleportToFarm()
    local character = player.Character
    if not character then return end
    local hrp = character:FindFirstChild("HumanoidRootPart")
    if not hrp then return end
    hrp.CFrame = CFrame.new(FARM_POSITION)
    task.wait(1)
end

-- ============================================
-- COW FINDER (1.5s CACHE)
-- ============================================
local CowCache = {}
local NextScan = 0

local function RefreshCowCache()
    table.clear(CowCache)
    local region = Region3.new(
        FARM_POSITION - Vector3.new(SEARCH_RADIUS, 50, SEARCH_RADIUS),
        FARM_POSITION + Vector3.new(SEARCH_RADIUS, 50, SEARCH_RADIUS)
    )
    local parts = Workspace:FindPartsInRegion3(region, nil, 500)
    for _, part in ipairs(parts) do
        local name = part.Name:lower()
        if name:find("sapi") or name:find("cow") then
            table.insert(CowCache, part)
        end
    end
end

local function GetNearestCow()
    local character = player.Character
    if not character then return nil end
    local hrp = character:FindFirstChild("HumanoidRootPart")
    if not hrp then return nil end
    if tick() > NextScan then
        RefreshCowCache()
        NextScan = tick() + 1.5
    end
    local nearest, closest = nil, math.huge
    for _, part in ipairs(CowCache) do
        if part and part.Parent then
            local dist = (hrp.Position - part.Position).Magnitude
            if dist < closest and dist <= SEARCH_RADIUS then
                closest = dist
                nearest = part
            end
        end
    end
    return nearest
end

-- ============================================
-- MOVE TO COW
-- ============================================
local function moveToCow(cowPart)
    local character = player.Character
    if not character then return false end
    local hrp = character:FindFirstChild("HumanoidRootPart")
    if not hrp or not cowPart then return false end
    local targetPos = cowPart.Position + Vector3.new(0, 0, 2.5)
    local distance = (hrp.Position - targetPos).Magnitude
    if distance > 2 then
        hrp.CFrame = CFrame.new(targetPos, cowPart.Position)
    end
    return true
end

-- ============================================
-- MINIGAME HANDLER (EVENT-DRIVEN, 3-STATE)
-- ============================================
local MinigameHandler = {
    State = "IDLE",       -- IDLE, DETECTED, EXECUTING
    PendingKeys = {},
    CurrentIndex = 0,
    LastPress = 0,
    CurrentMode = "",
    ProcessedTexts = {},
    Stats = {Total = 0, Success = 0, Failed = 0}
}

function MinigameHandler:Reset()
    self.State = "IDLE"
    self.PendingKeys = {}
    self.CurrentIndex = 0
    self.LastPress = 0
    self.CurrentMode = ""
end

function MinigameHandler:Execute()
    if self.State ~= "DETECTED" then return end
    self.State = "EXECUTING"
    self.Stats.Total += 1
    local startTime = tick()

    while self.State == "EXECUTING" do
        local now = tick()
        if now - self.LastPress < KEY_DELAY then
            task.wait()
            continue
        end

        local key = self.PendingKeys[self.CurrentIndex + 1]
        if not key then
            self.Stats.Success += 1
            self:Reset()
            return
        end

        if Enum.KeyCode[key] then
            self.CurrentIndex += 1
            self.LastPress = now
            pressKey(Enum.KeyCode[key])
        end

        if now - startTime > 2 then
            self.Stats.Failed += 1
            self:Reset()
            return
        end

        task.wait()
    end
end

-- Parse text and trigger minigame instantly
local function ProcessMinigameText(obj)
    if not obj:IsA("TextLabel") then return end
    if not obj.Visible then return end
    if obj.Text == "" then return end

    local text = cleanText(obj.Text)
    if text == "" then return end

    -- Duplicate filter (0.01s)
    if MinigameHandler.ProcessedTexts[text] and tick() - MinigameHandler.ProcessedTexts[text] < 0.01 then
        return
    end
    MinigameHandler.ProcessedTexts[text] = tick()

    -- Only process if IDLE
    if MinigameHandler.State ~= "IDLE" then return end

    local detected = nil

    -- Mode URUTAN
    if text:find("URUTAN") then
        local result = ""
        for letter in text:gmatch("%u") do
            if Enum.KeyCode[letter] then
                result = result .. letter
            end
        end
        result = result:gsub("URUTAN", "")
        if #result > 0 then
            detected = {Mode = "URUTAN", Keys = {}}
            for i = 1, #result do
                table.insert(detected.Keys, result:sub(i, i))
            end
        end

    -- Mode GASPOL
    elseif text:lower():find("gaspol") then
        local key = text:match("%[([A-Z])%]")
        local amount = tonumber(text:match("x(%d+)"))
        if key and amount and Enum.KeyCode[key] then
            amount = math.clamp(amount, 1, 30)
            detected = {Mode = "GASPOL", Keys = {}}
            for i = 1, amount do
                table.insert(detected.Keys, key)
            end
        end

    -- Mode PENCET
    elseif text:lower():find("pencet") then
        local key = text:match("%[([A-Z])%]")
        if key and Enum.KeyCode[key] then
            detected = {Mode = "PENCET", Keys = {key}}
        end
    end

    if detected then
        MinigameHandler.State = "DETECTED"
        MinigameHandler.PendingKeys = detected.Keys
        MinigameHandler.CurrentMode = detected.Mode
        MinigameHandler.CurrentIndex = 0
        MinigameHandler.LastPress = 0
        task.spawn(function()
            MinigameHandler:Execute()
        end)
    end
end

-- Setup event listeners for instant detection
local function SetupMinigameListeners()
    for _, v in ipairs(gui:GetDescendants()) do
        if v:IsA("TextLabel") then
            v:GetPropertyChangedSignal("Text"):Connect(function()
                ProcessMinigameText(v)
            end)
            v:GetPropertyChangedSignal("Visible"):Connect(function()
                if v.Visible then ProcessMinigameText(v) end
            end)
        end
    end
    gui.DescendantAdded:Connect(function(child)
        if child:IsA("TextLabel") then
            child:GetPropertyChangedSignal("Text"):Connect(function()
                ProcessMinigameText(child)
            end)
            child:GetPropertyChangedSignal("Visible"):Connect(function()
                if child.Visible then ProcessMinigameText(child) end
            end)
        end
    end)
end

-- Cleanup processed texts
task.spawn(function()
    while true do
        task.wait(5)
        local now = tick()
        for txt, time in pairs(MinigameHandler.ProcessedTexts) do
            if now - time > 2 then
                MinigameHandler.ProcessedTexts[txt] = nil
            end
        end
    end
end)

-- Minigame status display
task.spawn(function()
    while true do
        task.wait(0.15)
        if MinigameHandler.State ~= "IDLE" then
            SafeUI(string.format("Minigame %s | %s | %d/%d",
                MinigameHandler.CurrentMode,
                MinigameHandler.State,
                MinigameHandler.CurrentIndex,
                #MinigameHandler.PendingKeys
            ))
        end
    end
end)

-- ============================================
-- MASJID SYSTEM
-- ============================================
local MasjidCooldown = false

local function Teleport(targetCFrame, placeName)
    local character = player.Character or player.CharacterAdded:Wait()
    local hrp = character:WaitForChild("HumanoidRootPart", 10)
    local humanoid = character:WaitForChild("Humanoid", 10)
    if not hrp or not humanoid then return end

    local seat = humanoid.SeatPart
    local vehicle = seat and seat:FindFirstAncestorOfClass("Model")

    if vehicle and vehicle.PrimaryPart then
        local offset = vehicle.PrimaryPart.CFrame:ToObjectSpace(hrp.CFrame)
        vehicle:SetPrimaryPartCFrame(targetCFrame)
        task.wait()
        hrp.CFrame = vehicle.PrimaryPart.CFrame * offset
    else
        humanoid:ChangeState(Enum.HumanoidStateType.GettingUp)
        hrp.CFrame = targetCFrame
    end

    hrp.AssemblyLinearVelocity = Vector3.zero
    hrp.AssemblyAngularVelocity = Vector3.zero

    SafeUI("TELEPORT KE " .. placeName)
    Rayfield:Notify({
        Title = "Teleport",
        Content = "Berhasil ke " .. placeName,
        Duration = 2,
        Image = "map-pinned"
    })
end

local function Countdown(seconds, label)
    for i = seconds, 1, -1 do
        local progress = (seconds - i) / seconds
        local barLength = 20
        local filled = math.floor(progress * barLength)
        local bar = string.rep("█", filled) .. string.rep("░", barLength - filled)
        SafeUI(label .. "\n[" .. bar .. "] " .. math.floor(progress * 100) .. "% (" .. i .. "s)")
        task.wait(1)
    end
end

local function GetMasjidTarget()
    local pg = player:FindFirstChild("PlayerGui")
    if not pg then return nil end

    local masjid, studs = false, ""
    for _, v in ipairs(pg:GetDescendants()) do
        if v:IsA("TextLabel") then
            local t = v.Text
            if t and t ~= "" then
                t = string.lower(t)
                if not masjid and string.find(t, "masjid") then
                    masjid = true
                end
                if studs == "" and string.find(t, "studs") then
                    studs = t
                end
            end
        end
    end

    if not masjid then return nil end
    SafeUI("MASJID TERDETEKSI\nDISTANCE: " .. studs)

    if string.find(studs, "3%.0k") then
        return {Name = "MASJID 3K", Delay = 15, CFrame = CFrame.new(-4035.952148, -79.151535, -26729.810547)}
    end
    if string.find(studs, "9%.3k") then
        return {Name = "MASJID 9K", Delay = 25, CFrame = CFrame.new(-2842.638672, -38.547020, -14443.070312)}
    end
    if string.find(studs, "23%.8k") then
        return {Name = "MASJID 23K", Delay = 50, CFrame = CFrame.new(60.027203, 3.475821, -225.967621)}
    end
    return nil
end

-- ============================================
-- VEHICLE CHECK
-- ============================================
local function IsInVehicle()
    local character = player.Character
    if not character then return false end
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    if not humanoid then return false end
    return humanoid.SeatPart ~= nil
end

-- ============================================
-- MASJID DETECTION THREAD (INDEPENDENT)
-- ============================================
task.spawn(function()
    while true do
        task.wait(0.5)
        if not AutoSapi then continue end
        if MasjidCooldown then continue end

        local target = GetMasjidTarget()
        if target then
            MasjidCooldown = true
            SafeUI("MASJID TERDETEKSI: " .. target.Name)
            Rayfield:Notify({
                Title = "AUTO MASJID",
                Content = "Menuju " .. target.Name,
                Duration = 3,
                Image = "map"
            })

            Countdown(target.Delay, "MENUJU " .. target.Name)
            Teleport(target.CFrame, target.Name)
            SafeUI("SAMPAI: " .. target.Name .. "\nRESUME FARMING...")
            task.wait(2)

            task.spawn(function()
                task.wait(30)
                MasjidCooldown = false
            end)
        end
    end
end)

-- ============================================
-- MAIN FARMING LOOP
-- ============================================
local function startAutoCow()
    task.spawn(function()
        if not IsInVehicle() then
            teleportToFarm()
            task.wait(1)
        end

        while AutoSapi do
            -- Skip if in vehicle
            if IsInVehicle() then
                SafeUI("DI DALAM KENDARAAN\nTOGGLE ON | MASJID SCAN AKTIF")
                task.wait(0.5)
                continue
            end

            -- Skip if minigame active
            if MinigameHandler.State ~= "IDLE" then
                task.wait(0.1)
                continue
            end

            local cow = GetNearestCow()
            if not cow then
                SafeUI("Mencari sapi...")
                task.wait(0.5)
                continue
            end

            local success = moveToCow(cow)
            if success then
                task.wait(0.2)
                pressKey(Enum.KeyCode.E)
                -- Wait for minigame to finish (max 3s)
                local waitStart = tick()
                while MinigameHandler.State ~= "IDLE" and tick() - waitStart < 3 do
                    task.wait(0.1)
                end
            end

            task.wait(0.1)
        end
    end)
end
